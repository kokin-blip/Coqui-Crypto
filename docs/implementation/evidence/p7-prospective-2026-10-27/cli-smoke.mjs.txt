const cli=(action,extra=[])=>spawnSync(process.execPath,['scripts/research-capital-prospective.mjs',action,`--workspace=${root}`,`--database=${database}`,'--profile-id=main',...extra],{cwd:root,encoding:'utf8'}),check=(name,ok)=>{checks.push({name,passed:ok});if(!ok)throw new Error(name);};

 const design=core.identifiedBudgetFact({...prior,runtime,workspaceId:core.capitalHash({workspace:root}),startMs:start,developmentEndMs:prior.developmentEndMs+shift,finalStartMs:prior.finalStartMs+shift,endMs:prior.endMs+shift,foldEndsMs:prior.foldEndsMs.map(t=>t+shift),arms:prior.arms.map(arm=>{const changed={...arm,signal:{...arm.signal,anchorMs:arm.signal.anchorMs+shift}};return {...changed,strategyHash:core.prospectiveStrategyHash(changed)};}),trialHistory:{...prior.trialHistory,registryHash:identity.registryHash}}),request={commandId:randomUUID(),design},requestFile=join(folder,'request.json'),approvalFile=join(folder,'approval.json');writeFileSync(requestFile,JSON.stringify(request));const flags=[`--profile-manifest=${manifest}`,'--host-id=synthetic-cli-host',`--request=${requestFile}`];
 const prepared=cli('register',[...flags,'--prepare']);check('prepare succeeds without registration',prepared.status===0);db=storage.openDatabase(database,{skipMigrations:true});check('prepare does not append',db.prepare('SELECT COUNT(*) n FROM prospective_registrations_v1').get().n===0);db.close();const p=JSON.parse(prepared.stdout);writeFileSync(approvalFile,JSON.stringify({action:'register',requestHash:p.requestHash,designId:design.id,approvalHash:core.capitalHash('synthetic-only-owner'),reviewerRef:'synthetic-independent-review',registryCustodyRef:'synthetic-custody',exposureInventoryHash:p.exposureInventoryHash,authoritativeHistoryReviewed:true}));

 for(const action of ['inspect','coverage','development','registry'])check(`${action} local read`,cli(action,[`--registration-id=${r.registrationId}`]).status===0);

 console.log(JSON.stringify({synthetic:true,realProviders:false,checks},null,2));
}finally{rmSync(folder,{recursive:true,force:true});}
